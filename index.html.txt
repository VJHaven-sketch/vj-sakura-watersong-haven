<script>
async function loadData() {
  document.getElementById("status").innerText = "Loading...";

  const url = "https://docs.google.com/spreadsheets/d/e/2PACX-1vRrEuqpj3R8m7TFGabeVBTC_wwTgiBNmybVxGqcvd0ygZc46meLdmfqO83Z_2hJKy63CJM_JjB6jaCQ/pub?gid=2045933570&single=true&output=csv";

  try {
    const res = await fetch(url);
    const text = await res.text();

    const rows = text.split("\n").slice(1);

    const latest = rows[rows.length - 1].split(",");

    document.getElementById("data").innerHTML = `
      <h3>Latest Results</h3>
      <p>pH: ${latest[2]}</p>
      <p>KH: ${latest[5]}</p>
      <p>Ammonia: ${latest[8]}</p>
    `;

    document.getElementById("status").innerText = "Loaded ✔";
  } catch (e) {
    document.getElementById("status").innerText = "Error loading data";
  }
}
</script>
